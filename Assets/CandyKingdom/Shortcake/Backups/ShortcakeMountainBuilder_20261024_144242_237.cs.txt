 static Mesh TierMesh(string name,float radius,float height){
  var v=new List<Vector3>();var t=new List<int>();var uv=new List<Vector2>();int rings=12,sides=96;
  for(int j=0;j<=rings;j++){float u=j/(float)rings;float profile=u<.0834f?u/.0834f:u>.9166f?(1-u)/.0834f:1;
   float y=u<.0834f?0:u>.9166f?height:height*(u-.0834f)/.8332f;
   for(int k=0;k<=sides;k++){float a=k*Mathf.PI*2/sides;float r=radius*profile*(1+.014f*Mathf.Sin(a*5)+.009f*Mathf.Cos(a*9+1));v.Add(new Vector3(Mathf.Cos(a)*r,y,Mathf.Sin(a)*r));uv.Add(j==0||j>=11?new Vector2(Mathf.Cos(a)*profile*.5f+.5f,Mathf.Sin(a)*profile*.5f+.5f):new Vector2(k/12f,u));}}
  for(int j=0;j<rings;j++)for(int k=0;k<sides;k++){int a=j*(sides+1)+k,b=a+sides+1;t.Add(a);t.Add(b);t.Add(a+1);t.Add(a+1);t.Add(b);t.Add(b+1);}return Store(name,v,t,uv);
 }

  var v=new List<Vector3>();var t=new List<int>();var uv=new List<Vector2>();int sides=128;
  for(int j=0;j<3;j++)for(int k=0;k<=sides;k++){float a=k*Mathf.PI*2/sides;float r=radius*(1+.014f*Mathf.Sin(a*5)+.009f*Mathf.Cos(a*9+1))+.035f;float drop=.28f+.55f*Mathf.Pow(.5f+.5f*Mathf.Sin(a*11+.8f),4);v.Add(new Vector3(Mathf.Cos(a)*r,height-(j==0?0:j==1?.16f:drop),Mathf.Sin(a)*r));uv.Add(new Vector2(k/(float)sides,j*.5f));}
  for(int j=0;j<2;j++)for(int k=0;k<sides;k++){int a=j*(sides+1)+k,b=a+sides+1;t.Add(a);t.Add(a+1);t.Add(b);t.Add(a+1);t.Add(b+1);t.Add(b);}
  MeshObject("DrippingWhippedCream",Store(name+"_Drips",v,t,uv),bottom,Vector3.one,cream,g.transform);
  int pearls=Mathf.RoundToInt(radius*13);for(int k=0;k<pearls;k++){float a=k*Mathf.PI*2/pearls;Ball("RoundedIcingEdge",bottom+new Vector3(Mathf.Cos(a)*radius,height-.06f,Mathf.Sin(a)*radius),new Vector3(.54f,.27f,.54f),cream,g.transform);}
 }
 static Mesh BerryMesh(){
  var v=new List<Vector3>();var t=new List<int>();var uv=new List<Vector2>();
  for(int j=0;j<=24;j++){float u=j/24f;float r=.5f*Mathf.Pow(Mathf.Sin(u*Mathf.PI),.65f)*(.65f+.35f*u);for(int k=0;k<=48;k++){float a=k*Mathf.PI*2/48;v.Add(new Vector3(Mathf.Cos(a)*r,u,Mathf.Sin(a)*r));uv.Add(new Vector2(k/48f,u));}}
  for(int j=0;j<24;j++)for(int k=0;k<48;k++){int a=j*49+k,b=a+49;t.Add(a);t.Add(b);t.Add(a+1);t.Add(a+1);t.Add(b);t.Add(b+1);}return Store("Strawberry",v,t,uv);
 }

 static void Ramp(string name,Vector3 a,Vector3 b,float width,Material mat,Transform parent){
  var direction=Vector3.ProjectOnPlane(b-a,Vector3.up).normalized;var side=Vector3.Cross(Vector3.up,direction)*width*.5f;
  var v=new List<Vector3>{a-side,a+side,b+side,b-side,a-side-Vector3.up*.30f,a+side-Vector3.up*.30f,b+side-Vector3.up*.30f,b-side-Vector3.up*.30f};
  int[] tri={0,3,1,1,3,2,4,5,7,5,6,7,0,1,4,1,5,4,1,2,5,2,6,5,2,3,6,3,7,6,3,0,7,0,4,7};
  var uv=new List<Vector2>();foreach(var p in v)uv.Add(new Vector2(p.x*.5f,p.z*.5f));MeshObject(name,Store(name,v,new List<int>(tri),uv),Vector3.zero,Vector3.one,mat,parent,true);
 }

 static void Backup(string scene){

  File.Copy("Assets/CandyKingdom/Shortcake/Editor/ShortcakeMountainBuilder.cs",Root+"/Backups/ShortcakeMountainBuilder_"+stamp+".cs.txt");
  foreach(var path in Directory.GetFiles("Assets/CandyKingdom/ArtUpgrade/Editor","*.cs"))File.Copy(path,Root+"/Backups/"+Path.GetFileName(path)+"_"+stamp+".txt");
 }
 
 [MenuItem("Tools/Candy Kingdom/Rebuild Shortcake Visual Batches")]
 public static void Optimize(){
  if(EditorApplication.isPlayingOrWillChangePlaymode||SceneManager.GetActiveScene().path!=ScenePath)throw new Exception("Open two-route scene in Edit mode.");
  var root=GameObject.Find("ShortcakeAdventureGenerated").transform;var previous=root.Find("BatchedShortcakeVisuals");if(previous)UnityEngine.Object.DestroyImmediate(previous.gameObject);
  var groups=new Dictionary<Material,List<CombineInstance>>();var originals=new List<MeshRenderer>();
  foreach(var mf in root.GetComponentsInChildren<MeshFilter>()){var mr=mf.GetComponent<MeshRenderer>();if(!mr||!mf.sharedMesh||mr.sharedMaterials.Length!=1||!mr.sharedMaterial)continue;var material=mr.sharedMaterial;if(!groups.ContainsKey(material))groups[material]=new List<CombineInstance>();groups[material].Add(new CombineInstance{mesh=mf.sharedMesh,transform=root.worldToLocalMatrix*mf.transform.localToWorldMatrix});originals.Add(mr);}
  var combined=Group("BatchedShortcakeVisuals",root,Vector3.zero);
  foreach(var pair in groups){string path=Root+"/Meshes/RouteBatch_"+pair.Key.name+".asset";var mesh=AssetDatabase.LoadAssetAtPath<Mesh>(path);if(!mesh){mesh=new Mesh();mesh.name="RouteBatch_"+pair.Key.name;AssetDatabase.CreateAsset(mesh,path);}else mesh.Clear();mesh.indexFormat=UnityEngine.Rendering.IndexFormat.UInt32;mesh.CombineMeshes(pair.Value.ToArray(),true,true,false);mesh.RecalculateBounds();EditorUtility.SetDirty(mesh);AssetDatabase.SaveAssetIfDirty(mesh);MeshObject(pair.Key.name,mesh,Vector3.zero,Vector3.one,pair.Key,combined.transform);}

 }
